# nvinfer config for PeopleNet (FP16, ResNet34, 3 classes: Person/Bag/Face).
# Used by /usr/local/bin/app.py via the nvinfer config-file-path property.
#
# FP16 (network-mode=2) chosen over INT8 because the engine builds in
# ~60 seconds on Orin Nano instead of 3–5 minutes. Inference is ~10–15%
# slower than INT8 but still real-time at 720p; accuracy is essentially
# identical for the demo workload.

[property]
gpu-id=0
net-scale-factor=0.0039215697906911373

# Model file. Shipped read-only in the vision-models sysext.
onnx-file=/usr/lib/nvidia-deepstream/models/peoplenet/resnet34_peoplenet_int8.onnx
labelfile-path=/usr/lib/nvidia-deepstream/models/peoplenet/labels.txt

# Prebuilt FP16 engine, shipped read-only in the vision-engines sysext.
# nvinfer mmaps the engine from this path on load — it never compiles or
# writes, so the engine can live on read-only /usr instead of being staged to
# /var. (nvinfer only writes an engine if it has to BUILD one from the ONNX,
# which on this immutable layout would fail on the read-only FS — which is why
# every target ships a prebuilt engine; there is no on-device compile.) The
# filename must match nvinfer's convention:
# `<onnx-basename>_b<batch>_gpu<id>_<mode>.engine`.
model-engine-file=/usr/lib/nvidia-deepstream/engines/peoplenet/resnet34_peoplenet_int8.onnx_b1_gpu0_fp16.engine

batch-size=1
# 0 = FP32, 1 = INT8, 2 = FP16
network-mode=2
num-detected-classes=3
interval=0
gie-unique-id=1
process-mode=1
model-color-format=0
# Intentionally omitted: nvinfer + TensorRT 10.x read IO tensor names directly
# from the ONNX graph. Hard-coding the old TAO TF1 names (output_bbox/BiasAdd,
# output_cov/Sigmoid) makes getTensorIOMode() reject them at engine load, and
# the pipeline silently stalls: nvinfer never binds its outputs, no buffers
# flow, and v4l2src eventually errors out with `not-linked (-1)` minutes later.
# Leave unset and let nvinfer auto-discover the outputs.
#output-blob-names=output_bbox/BiasAdd;output_cov/Sigmoid
cluster-mode=2

[class-attrs-all]
pre-cluster-threshold=0.4
topk=20
nms-iou-threshold=0.5
